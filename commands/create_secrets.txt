# generate SSH secret for github
kubectl create secret generic git-ssh-secret \
  -n airflow \
  --from-file=gitSshKey=/home/airflow/.ssh/id_ed25519 \
  --from-file=knownHosts=./known_hosts \
  --dry-run=client -o yaml | kubectl apply -f -


# generate secret for docker hub credentials  
kubectl create secret generic registry-credentials \
  -n airflow \
  --from-literal=REGISTRY_USER="<dockerhub account>" \
  --from-literal=REGISTRY_PASS="<password or personal access token">
